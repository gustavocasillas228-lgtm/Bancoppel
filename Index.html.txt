    <script>
        // Catálogo de Productos
        const products = [
            { id: 1, name: "Gorra Urbana Snapback", price: 350, cat: "moda", img: "https://images.unsplash.com/photo-1588850561407-ed78c282e89b?w=400" },
            { id: 2, name: "Audífonos Bluetooth Pro", price: 650, cat: "tech", img: "https://images.unsplash.com/photo-1505740420928-5e560c06d30e?w=400" },
            { id: 3, name: "Lentes Fotocromáticos UV400", price: 420, cat: "accesorios", img: "https://images.unsplash.com/photo-1572635196237-14b3f281503f?w=400" },
            { id: 4, name: "Sudadera Hoodie Oversize", price: 550, cat: "moda", img: "https://images.unsplash.com/photo-1556905055-8f358a7a47b2?w=400" },
            { id: 5, name: "Funda Uso Rudo Anti-Impacto", price: 220, cat: "tech", img: "https://images.unsplash.com/photo-1541877944-ac82a091518a?w=400" },
            { id: 6, name: "Cadena + Pulsera Color Plata", price: 480, cat: "accesorios", img: "https://images.unsplash.com/photo-1599643478518-a784e5dc4c8f?w=400" }
        ];

        let cart = [];
        let currentUser = null;

        // Renderizar Productos
        function renderProducts(items) {
            const grid = document.getElementById('productsGrid');
            grid.innerHTML = '';
            
            items.forEach(product => {
                grid.innerHTML += `
                    <div class="product-card">
                        <img src="${product.img}" alt="${product.name}" class="product-img">
                        <div class="product-body">
                            <span class="product-cat">${product.cat}</span>
                            <h3 class="product-title">${product.name}</h3>
                            <div class="product-price">$${product.price}.00 MXN</div>
                            <button class="btn-add" onclick="addToCart(${product.id})">
                                <i class="fas fa-plus"></i> Agregar al carrito
                            </button>
                        </div>
                    </div>
                `;
            });
        }

        // Agregar al Carrito
        function addToCart(id) {
            const product = products.find(p => p.id === id);
            cart.push(product);
            updateCartUI();
        }

        // Actualizar UI Carrito
        function updateCartUI() {
            document.getElementById('cartBadge').innerText = cart.length;
            const container = document.getElementById('cartItems');
            container.innerHTML = '';
            
            let total = 0;
            cart.forEach((item, index) => {
                total += item.price;
                container.innerHTML += `
                    <div class="cart-item">
                        <img src="${item.img}">
                        <div class="cart-item-info">
                            <div class="cart-item-title">${item.name}</div>
                            <div class="cart-item-price">$${item.price}.00</div>
                        </div>
                        <i class="fas fa-trash" style="color:#ef4444; cursor:pointer;" onclick="removeFromCart(${index})"></i>
                    </div>
                `;
            });

            document.getElementById('cartTotalSum').innerText = `$${total}.00 MXN`;
        }

        function removeFromCart(index) {
            cart.splice(index, 1);
            updateCartUI();
        }

        // Filtros y Búsqueda
        document.getElementById('searchInput').addEventListener('input', (e) => {
            const term = e.target.value.toLowerCase();
            const filtered = products.filter(p => p.name.toLowerCase().includes(term));
            renderProducts(filtered);
        });

        document.getElementById('categoriesFilter').addEventListener('click', (e) => {
            if (e.target.classList.contains('filter-btn')) {
                document.querySelectorAll('.filter-btn').forEach(btn => btn.classList.remove('active'));
                e.target.classList.add('active');
                
                const cat = e.target.getAttribute('data-cat');
                if (cat === 'all') renderProducts(products);
                else renderProducts(products.filter(p => p.cat === cat));
            }
        });

        // Modales y Drawer
        const cartDrawer = document.getElementById('cartDrawer');
        const backdrop = document.getElementById('backdrop');
        const authModal = document.getElementById('authModal');

        document.getElementById('cartBtn').onclick = () => { cartDrawer.classList.add('open'); backdrop.classList.add('open'); };
        document.getElementById('closeCart').onclick = () => { cartDrawer.classList.remove('open'); backdrop.classList.remove('open'); };
        backdrop.onclick = () => { cartDrawer.classList.remove('open'); backdrop.classList.remove('open'); authModal.classList.remove('open'); };

        document.getElementById('authBtn').onclick = () => { authModal.classList.add('open'); };
        document.getElementById('closeAuth').onclick = () => { authModal.classList.remove('open'); };

        // Login Form
        document.getElementById('loginForm').onsubmit = (e) => {
            e.preventDefault();
            currentUser = document.getElementById('usernameInput').value;
            document.getElementById('userText').innerText = currentUser;
            authModal.classList.remove('open');
        };

        // Checkout WhatsApp
        document.getElementById('checkoutBtn').onclick = () => {
            if(cart.length === 0) return alert('El carrito está vacío');
            
            const telefono = "526640000000"; // Agrega tu número de WhatsApp
            let mensaje = `*Nuevo Pedido de Bancoppel*%0A`;
            if(currentUser) mensaje += `*Cliente:* ${currentUser}%0A`;
            mensaje += `%0A*Productos:*%0A`;
            
            let total = 0;
            cart.forEach(item => {
                mensaje += `- ${item.name} ($${item.price}.00 MXN)%0A`;
                total += item.price;
            });
            
            mensaje += `%0A*Total a pagar:* $${total}.00 MXN`;
            window.open(`https://wa.me/${telefono}?text=${mensaje}`, '_blank');
        };

        // Inicializar
        renderProducts(products);
    </script>
